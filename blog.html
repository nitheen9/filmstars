async function loadPosts() { const page = getPage(); /* * Read label from URL. * * Example: * * /blog?label=Tollywood%20Actress */ const params = new URLSearchParams( window.location.search ); const selectedLabel = ( params.get("label") || "" ).trim(); /* * Calculate Blogger start index. */ const startIndex = ( (page - 1) * PER_PAGE ) + 1; postsContainer.innerHTML = `
Loading posts...
`; try { /* * ================================================== * BUILD API URL * ================================================== */ const apiUrl = new URL( API, window.location.origin ); apiUrl.searchParams.set( "start-index", startIndex ); apiUrl.searchParams.set( "max-results", PER_PAGE ); /* * IMPORTANT: * * Send the label to the server. * * Do NOT download 150 posts and filter * them in JavaScript. */ if ( selectedLabel ) { apiUrl.searchParams.set( "label", selectedLabel ); } /* * ================================================== * FETCH * ================================================== */ const response = await fetch( apiUrl.toString(), { cache: "no-store" } ); if ( !response.ok ) { throw new Error( `HTTP ${response.status}` ); } const data = await response.json(); if ( !data.success ) { throw new Error( data.error || "Unable to load posts" ); } const posts = Array.isArray( data.posts ) ? data.posts : []; /* * ================================================== * LABEL HEADER * ================================================== */ const normalHeader = document.getElementById( "normalHeader" ); const labelHeader = document.getElementById( "labelHeader" ); const labelTitle = document.getElementById( "labelTitle" ); const labelDescription = document.getElementById( "labelDescription" ); if ( selectedLabel ) { if (normalHeader) { normalHeader.style.display = "none"; } if (labelHeader) { labelHeader.style.display = "block"; } if (labelTitle) { labelTitle.textContent = `Label: ${selectedLabel}`; } if (labelDescription) { labelDescription.textContent = `${data.total} post${ Number(data.total) === 1 ? "" : "s" } found with this label.`; } } else { if (normalHeader) { normalHeader.style.display = "block"; } if (labelHeader) { labelHeader.style.display = "none"; } } /* * ================================================== * NO POSTS * ================================================== */ if ( posts.length === 0 ) { postsContainer.innerHTML = `

No Posts Found

${ selectedLabel ? `

No posts found with the label:

${escapeHtml( selectedLabel )} ` : `

No posts found.

` }

← View All Posts
`; paginationContainer.innerHTML = ""; return; } /* * ================================================== * DISPLAY POSTS * ================================================== */ postsContainer.innerHTML = posts .map( createCard ) .join(""); /* * ================================================== * PAGINATION * ================================================== * * data.total is the TOTAL number of Blogger * posts, not just the 20 currently displayed. */ createPagination( page, Number(data.total) || posts.length ); } catch (error) { console.error( "Filmstars Blogger error:", error ); postsContainer.innerHTML = `
Unable to load posts right now.

Please try again in a few seconds.

`; paginationContainer.innerHTML = ""; } }
